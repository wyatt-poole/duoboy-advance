cmake_minimum_required(VERSION 3.22)
project(gbads C)
set(MGBA ${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/mgba)
set(BUILD_STATIC ON CACHE BOOL "" FORCE)
set(BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(M_CORE_GB OFF CACHE BOOL "" FORCE)
foreach(opt USE_LUA USE_JSON_C USE_FREETYPE USE_LZMA USE_ELF USE_ZLIB USE_PNG USE_LIBZIP USE_MINIZIP USE_SQLITE3 USE_FFMPEG USE_EDITLINE USE_DEBUGGERS USE_GDB_STUB USE_DISCORD_RPC ENABLE_SCRIPTING BUILD_GL BUILD_GLES2 BUILD_GLES3 USE_EPOXY)
  set(${opt} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory(${MGBA} mgba EXCLUDE_FROM_ALL)
set(RC ${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/rcheevos)
file(GLOB RC_SRC ${RC}/src/api/*.c ${RC}/src/client/rc_client.c ${RC}/src/hash/rc_hash.c ${RC}/src/hash/rc_hash_rom.c ${RC}/src/runtime/*.c ${RC}/src/util/*.c)
add_library(gbads SHARED bridge.c ra.c ${MGBA}/src/platform/libretro/libretro-audio.c ${RC_SRC})
target_include_directories(gbads PRIVATE ${RC}/include)
# GBA is hashed as a plain ROM; skip disc/zip/encrypted hashers.
target_compile_definitions(gbads PRIVATE RC_CLIENT_SUPPORTS_HASH RC_HASH_NO_DISC RC_HASH_NO_ZIP RC_HASH_NO_ENCRYPTED)
target_include_directories(gbads PRIVATE ${MGBA}/src/platform/libretro ${MGBA}/include ${CMAKE_CURRENT_BINARY_DIR}/mgba/include)
target_link_libraries(gbads mgba log)
# mGBA's struct layouts depend on its private defines (e.g. ENABLE_DIRECTORIES); compile the bridge with the same set.
get_target_property(MGBA_DEFS mgba COMPILE_DEFINITIONS)
target_compile_definitions(gbads PRIVATE ${MGBA_DEFS})
